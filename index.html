// 🌍 OpenWeather API KEY const API_KEY = "9de43a81fe7ad37cf116c9b9defe5826"; // 🌤️ MAIN WEATHER FUNCTION function showSimpleWeather(city) { fetch(https://api.openweathermap.org/data/2.5/weather?q=${city},AL&units=metric&appid=${API_KEY}) .then(res => res.json()) .then(data => { if (!data || !data.main) return; // 🌡️ DATA let temp = data.main.temp; let wind = data.wind.speed; let weather = data.weather[0].description; // 💡 SUGGESTION SYSTEM let suggestion = ""; if (temp >= 30) { suggestion = "🔥 Very hot day — drink water or stay in shade!"; } else if (temp >= 22) { suggestion = "🌤️ Nice weather — perfect for walking!"; } else if (temp >= 15) { suggestion = "🌥️ Mild weather — light jacket recommended."; } else { suggestion = "🧥 Cold weather — wear warm clothes!"; } if (wind >= 8) { suggestion += " 🌬️ Windy — be careful outside!"; } // 📦 DISPLAY (you must have these elements in HTML) document.getElementById("temp").innerText = "🌡️ Temp: " + temp + "°C"; document.getElementById("wind").innerText = "🌬️ Wind: " + wind + " m/s"; document.getElementById("weather").innerText = "🌤️ " + weather; document.getElementById("advice").innerText = suggestion; }) .catch(err => { console.log("Weather error"); }); }
